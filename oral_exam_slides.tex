\section{Terminology and scope}

\begin{frame}{The MBSE problem and the standard solvers}
\small
For $N$ electrons in 2D ($d{=}2N$ configuration space):
\[
  \hat H \Psi = E\Psi,\qquad
  \hat H = \sum_i\bigl(-\tfrac12\nabla_i^2 + \tfrac12\omega^2 r_i^2\bigr)
         + \sum_{i<j}\tfrac{1}{r_{ij}}.
\]
\begin{tabular}{@{}ll@{}}
\paperA{HF}    & mean-field; one Slater determinant; misses correlation.\\
\paperA{CCSD}  & truncated cluster expansion; scales polynomially; fails in strong-correlation regimes (Wigner).\\
\paperA{FCI}   & exact in a basis; basis dimension grows combinatorially in $d$.\\
\paperA{DMC}   & stochastic projection $\propto e^{-\tau\hat H}$ onto the ground state; fixed-node bias.\\
\paperA{VMC}   & sample $|\Psi_\theta|^2$, evaluate $\langle\hat H\rangle$; minimize variationally.\\
\paperA{NQS}   & neural ansatz used inside VMC (Carleo--Troyer 2017 onward).
\end{tabular}

\vspace{4pt}
\begin{tcolorbox}[colback=ourwork!8,colframe=ourwork!60,boxsep=2pt]
\small \ours{What ML buys:} expressivity \emph{and} scaling --- FCI has the first, CCSD the second.
The bet is that a neural ansatz can have both.
\end{tcolorbox}
\end{frame}

% ============================================================
\begin{frame}{Our ansatz}
\small
\[
  \Psi_{\theta,\beta}(R) \;=\; \mathrm{SD}\bigl(\tilde R + \Delta_\beta(\tilde R)\bigr)\,
  \exp\!\Big(\underbrace{\textstyle\sum_{i<j}\!u_{\sigma_i\sigma_j}(\tilde r_{ij})}_{\text{analytic cusp}}
  + \underbrace{W_\theta(\tilde R)}_{\text{neural correlator}}\Big).
\]
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\textbf{Pieces.}
\begin{itemize}\setlength\itemsep{1pt}\small
\item \ours{Slater determinant}: antisymmetry.
\item \ours{Analytic cusp}: $1/r$ Kato condition; removes the singular part from $E_L$.
\item \ours{Neural correlator $W_\theta$}: three Deep-Sets-style branches ($\phi,\psi,\rho$).
\item \ours{Backflow $\Delta_\beta$}: Deep sets style / CTNN network; deforms the nodal surface.
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{Trap units} $\tilde r = \sqrt\omega\,r$: makes the harmonic length $\ell\!\to\!1$ across $\omega$.

\vspace{2pt}
\textbf{Why the determinant sees backflow but the correlator does not.} Keeps the cusp/Jastrow on the
unmodified coordinates --- prevents a self-reinforcing feedback into the local energy.

\vspace{2pt}
\textbf{Why three networks.} Particle-only Deep Sets cannot represent pair correlation
(Haas et al.); a pair branch is mandatory.
\end{column}
\end{columns}
\end{frame}

% ============================================================
\begin{frame}{CTNN architecture}
\small
\textbf{One round of message passing on the particle graph:}
\[
  \mathbf m_{ij} = \phi_\beta(\tilde{\mathbf x}_i,\tilde{\mathbf x}_j,\mathbf r_{ij},\ldots)\cdot w^{\rm spin}_{ij},
  \quad \mathbf m_i = \sum_{j\ne i}\mathbf m_{ij},\quad
  \delta\tilde{\mathbf x}_i = \psi_\beta(\tilde{\mathbf x}_i\|\mathbf m_i),
\]
\[
  \Delta_\beta \;=\; \tanh(\delta\tilde X)\cdot s_\beta,\qquad s_\beta = \mathrm{softplus}(s^{\rm raw}_\beta)>0,\quad
  \text{last layer of }\psi_\beta \text{ zero-initialized.}
\]
\begin{columns}[T,onlytextwidth]
\begin{column}{0.50\textwidth}
\textbf{Why these choices.}
\begin{itemize}\setlength\itemsep{1pt}\small
\item Permutation-invariant (sum aggregation).
\item Rotation-equivariant (vector messages).
\item $k{=}1$ round of MP $\Rightarrow$ up to \emph{three-body} correlations.
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\section{Paper A --- Modern Mathematics of Deep Learning}

\begin{frame}{Paper A --- the four organizing questions}
\paperA{Berner, Grohs, Kutyniok, Petersen (2021).}
\small
\[
  R(f_S) - R(f^\star) \;=\; \underbrace{R(f_S)-R(f_{\mathcal F})}_{\text{generalization}}
  + \underbrace{R(f_{\mathcal F})-R(f^\star)}_{\text{approximation}}.
\]
Classical bounds (VC / Rademacher) need $\dim(\mathcal F)\ll n$. Deep nets have parameters $\gg$ data, fit
the training set exactly, yet generalize. \emph{Classical bounds are vacuous.}

\vspace{4pt}
\begin{itemize}\setlength\itemsep{1pt}\small
\item \paperA{Q1} How do over-parameterized NNs \emph{generalize}?
\item \paperA{Q2} What role does \emph{depth} / width play in expressivity?
\item \paperA{Q3} How do NNs beat the \emph{curse of dimensionality}?
\item \paperA{Q4} Why does \emph{non-convex optimization} succeed?
\end{itemize}
\end{frame}

% ----------------- A Q1 ------------------------
\begin{frame}{\paperA{A--Q1}: Generalization}
\small
\textbf{The phenomenology is robust; no unified theory.}

\vspace{2pt}
\begin{itemize}\setlength\itemsep{2pt}\small
\item \paperA{Double descent} (Belkin et al.\ 2019): past the interpolation threshold, test error
falls a \emph{second} time. Empirical fact the classical U-curve cannot accommodate.
\item \paperA{NTK regime} (Jacot 2018, very wide nets): training linearizes into kernel regression
with the Neural Tangent Kernel
$\Theta(x,x')=\langle\nabla_\theta f(x),\nabla_\theta f(x')\rangle$ frozen at init; convergence set
by its spectrum.
\item \ours{For us:} the SR matrix $S=\mathrm{Cov}_{|\Psi|^2}[\partial_\theta\log|\Psi|]$ is exactly
this NTK (of $\log|\Psi|$) in parameter space.
\end{itemize}

\vspace{4pt}
\begin{tcolorbox}[colback=papA!5,colframe=papA!60,boxsep=2pt]
\small \textbf{Honest verdict (article, \S1.2):} robust phenomenology, no unified theory.
\end{tcolorbox}
\end{frame}

% ----------------- A Q2 ------------------------
\begin{frame}{\paperA{A--Q2}: Expressivity --- depth and width}
\small
\textbf{Universal approximation} (Cybenko 1989, Hornik 1991): one hidden layer with non-poly.\
activation is dense in $C(K)$ --- a \emph{topological} statement, no rate.

\vspace{4pt}
\textbf{Width vs depth (article example).} Approximating $\|x-y\|$ on $[0,1]^d\times[0,1]^d$ to error
$\varepsilon$:
\[
  \text{1 hidden layer: width } \Omega\!\bigl(2^{cd}\bigr),\qquad
  \text{multi-layer: width } \mathrm{poly}(d).
\]
Depth provides \emph{categorically} larger function classes at fixed budget.

\vspace{4pt}
\textbf{What width is for (article).} With $\mathrm{ReLU}$, width $\ge d_{\rm in}+1$ is necessary for
universality at all (Hanin--Sellke); width controls the dimension of the manifold along which the
function can vary smoothly. Depth controls compositionality; width controls cross-section.

\vspace{4pt}
\begin{tcolorbox}[colback=papA!5,colframe=papA!60,boxsep=2pt]
\small Depth is how NNs realize hierarchical / compositional structure that classical
(sparse-grid, spectral) methods cannot.
\end{tcolorbox}
\end{frame}

% ----------------- A Q3 ------------------------
\begin{frame}{\paperA{A--Q3}: Curse of dimensionality --- Barron, intrinsic $d$}
\small
\textbf{Curse:} approx.\ $C^s$ function on $\mathbb R^d$ to error $\varepsilon$ needs
$\sim \varepsilon^{-d/s}$ neurons --- exponential in $d$. ($s=$ smoothness order; the more
derivatives the target has, the faster classical methods converge.)

\vspace{4pt}
\textbf{Barron's theorem.} For $g$ with
$C_g=\!\int_{\mathbb R^d}\!\|\xi\|_2\,|\hat g(\xi)|\,d\xi<\infty$, a two-layer net with $n$ neurons satisfies
\[
  \inf_\theta\|\Phi_{(d,n,1),\varrho}(\cdot,\theta)-g\|_{L^2(\mu)}
   \;\le\; \frac{c\,C_g}{\sqrt n}.
\]
\centering\colorbox{papA!10}{\parbox{0.72\linewidth}{\centering\small Exponent \textbf{independent of $d$.}
Dimension hidden in $C_g$. Proof: Monte Carlo over $n$ random neurons.}}

\vspace{4pt}
\raggedright
\textbf{Connection to MBSE.} \paperA{HF / FCI} take a $d{=}40$ problem and expand against a basis
that grows combinatorially. \paperA{CCSD} keeps polynomial scaling by sacrificing expressivity ---
collapses in the Wigner regime. \ours{NN ansatz}: find the \emph{intrinsic} dimension.

\vspace{2pt}
\paperA{Note (\S1.4.3):} Coulomb / cusp parts of $\Psi$ are \emph{not} in Barron space; we
\emph{condition} the problem (cusp factor, soft features) to push the learned part into a
Barron-friendly class.
\end{frame}

% ----------------- A Q4 ------------------------
\begin{frame}{\paperA{A--Q4}: Optimization --- gradients, NTK, SR}
\small
\textbf{The puzzle.} The loss is non-convex with exponentially many critical points; worst-case
theory says SGD should fail. It doesn't. Why?

\vspace{4pt}
\begin{itemize}\setlength\itemsep{2pt}\small
\item \paperA{Landscape \& saddle escape}: critical points are dominated by saddles, not bad local
minima (Dauphin 2014); the stochastic noise in SGD kicks the iterate off saddles.
\item \paperA{Implicit regularization}: among the many interpolating solutions, (S)GD from small init
drifts to the min-norm / flat one $\Rightarrow$ good generalization for free.
\item \paperA{Adam}: per-coordinate adaptive step; diagonal preconditioner from the running second
moment of the gradient. Robust, axis-aligned curvature only.
\item \paperA{Natural gradient}: step $S^{-1}\nabla L$ with the Fisher matrix $S$.
\end{itemize}

\vspace{4pt}
\textbf{Architecture as optimization aid.}  CNN/GNN equivariance shrinks
the function class.
\end{frame}

\section{Paper B --- Numerical Analysis of PINNs}

\begin{frame}{Paper B --- spine, Q1--Q3 in one slide}
\paperB{De Ryck \& Mishra (2024).}
\small
\[
  \|u_\theta - u\|_{\mathcal X}
  \;\le\; C_{\rm stab}\bigl(\underbrace{\mathcal E_{\rm approx}}_{\text{Q1}}
  + \underbrace{\mathcal E_{\rm gen}}_{\text{Q3}}
  + \underbrace{\mathcal E_{\rm opt}}_{\text{Q4}}\bigr).
\]
\begin{itemize}\setlength\itemsep{1pt}\small
\item \paperB{Q1 --- approximation.} $\tanh$-nets reach Sobolev rate $\varepsilon^{-d/(k-j)}$;
this is just universal approximation in $H^k$. For us \paperA{Paper A} suffices.
\item \paperB{Q2 --- stability.} Assume a low generalization loss; \emph{then} low training loss bounds
the solution. \ours{For us:} zero-variance principle. $\mathrm{Var}[E_L]=0 \iff \Psi$ is an exact
eigenstate; both losses vanish together.
\item \paperB{Q3 --- generalization.} Concentration:
$\sup_\theta|L_{\rm train}-L_{\rm true}|\lesssim \mathrm{Lip}\sqrt{\log\mathcal N/n}$.
\ours{For us:} always evaluated by an independent heavy VMC.
\end{itemize}

\vspace{2pt}
\begin{tcolorbox}[colback=papB!5,colframe=papB!60,boxsep=2pt]
\small \textbf{Paper B's headline:} the binding bottleneck is \emph{training}, and it is governed by
the spectrum of the \emph{Hermitian square} $L^\ast L$.
\end{tcolorbox}
\end{frame}

% ----------------- B Q4 ------------------------
\begin{frame}{\paperB{B--Q4}: Training $=$ conditioning of $L^\ast L \circ TT^\ast$}
\small
Linearize GD around $\theta_0$: $q_i=\partial_{\theta_i}u_{\theta_0}$.
\[
  \underbrace{A_{ij}}_{\substack{p\times p,\ \text{param space}}} = \langle Lq_i,Lq_j\rangle
  \kappa(A)=\lambda_{\max}/\lambda_{\min}\ \text{ controls training cost.}
\]
\textbf{Theorem 7.10.} $\kappa(A) = \kappa(\mathbf A\circ TT^\ast)$
($\circ$ = operator composition), with $\mathbf A=L^\ast L$ \emph{the PDE operator on $L^2(\Omega)$}
, and $TT^\ast$ the \emph{NTK as an integral operator}. Composition
$=$ stiffness GD feels.

\vspace{4pt}
\textbf{Laplacian.} $L=-\Delta \Rightarrow L^\ast L=\Delta^2$; Fourier symbol $|k|^4 \Rightarrow
\kappa\sim(k_{\max}/k_{\min})^4$.
Supervised learning is the special case $L{=}\mathrm{Id}$, $\mathbf A{=}I$. All extra PINN
difficulty is in $L^\ast L$.

\vspace{4pt}
\textbf{Preconditioning:} choose $P$ with $PP^\top \approx (L^\ast L)^{-1}$.
\end{frame}

% ----------------- B Q4 / our negative result ------------------------
\begin{frame}{\ours{Catch-22:} $TT^\ast$-preconditioning is not enough when $L^\ast L$ is stiff}
\small
\textbf{Setting.} $N{=}6,\omega{=}0.5$, backflow ansatz, collocation training.
\textbf{17 interventions} (hard mining, smoothness penalty, hybrid loss, det filter, soft
Coulomb, kinetic clipping,Adam variants, \ldots): all give error in $[0.32,\,24.7]\%$.
\textbf{Same ansatz, VMC$+$SR}: $<0.01\%$.

\vspace{4pt}
\textbf{What helped.} Removing the Laplacian path from backflow gradients (REINFORCE only) while
keeping the kinetic term in the loss. $L^\ast L$ no longer couples to backflow parameters.

\vspace{2pt}
\textbf{Why the analytic cusp works.} Kato slope fixed analytically $\Rightarrow$ the network never
represents $1/r$ in $E_L$; a non-Barron feature taken out of the function class entirely.
\end{frame}

% ----------------- Three-body ----------
\begin{frame}{Three-body sensitivity --- what one extra round of MP buys}
\small
\textbf{Definition.} Bin by pair distance. Pairwise-only $f$ has zero intra-bin variance by
construction; the CTNN cell can have variance from beyond-pairwise structure (a third particle
changes the message to $i$ even when $r_{ij}$ is fixed). Ratio:
\[
  r(\omega) = \mathrm{Var}_{\rm intra}[\text{CTNN cell}]\,/\,\mathrm{Var}_{\rm intra}[\text{pairwise}].
\]
\textbf{Data ($N{=}6$).} $r(1.0)=2.03,\ r(0.1)=2.93,\ r(0.001)=1.05$.

\vspace{2pt}
\textbf{Interpretation.} At moderate-to-weak correlation the cell genuinely needs three-body content; in
deep Wigner the structure is set by the classical lattice (pairwise), capacity present but unused.

\vspace{2pt}
\begin{tcolorbox}[colback=ourwork!8,colframe=ourwork!60,boxsep=2pt]
\small \ours{Codimension-in-pairwise:} $r(\omega)\to 1$ when the optimal correlator lies in
$\mathcal F_0$; $r(\omega)>1$ otherwise. A variance ratio measuring the codimension of the
physical solution inside the architectural function class.
\end{tcolorbox}
\end{frame}

\begin{frame}{Synthesis}
\small
\begin{itemize}\setlength\itemsep{1pt}
\item \paperA{Paper A}: the manifold \emph{exists} --- Barron, depth, compositionality, symmetry
break the curse; architecture encodes the symmetry.
\item \paperB{Paper B}: the manifold is \emph{reachable} iff $\kappa(L^\ast L\circ TT^\ast)$ is
controlled --- nearly every successful trick is preconditioning.
\item \ours{Our PINN} sits at the intersection: conditioning (cusp, soft features, trap units, REINFORCE,
SR/Adam regime split) brings $r_{\rm eff}\le 3$ within reach for $d=40$.
\end{itemize}

\vspace{4pt}
\begin{tcolorbox}[colback=ourwork!8,colframe=ourwork!60,boxsep=2pt]
\small\textbf{Take-away.} The many-body wavefunction lives on a $1$--$3$-dimensional manifold that a
symmetry-equivariant, well-conditioned network can reach with $\sim 3\times 10^3$ MCMC samples per
SR step --- orders of magnitude better than basis blow-up implies.
\end{tcolorbox}
\end{frame}

\begin{frame}{Backup: NTK proof sketch and the linearization}
\small
$f_t(x)=f_{\theta_t}(x)$. Chain rule on $\dot\theta=-\nabla_\theta L$:
\[
  \dot f_t(x) \;=\; -\!\sum_{x'}\!\Theta_t(x,x')\bigl(f_t(x')-y(x')\bigr),\qquad
  \Theta_t(x,x')=\sum_p \partial_{\theta_p}\!f_t(x)\,\partial_{\theta_p}\!f_t(x').
\]
\textbf{Jacot.} As width $m\!\to\!\infty$ with appropriate scaling: parameters move only
$O(1/\sqrt m)$ while the function moves $O(1)$, so $\Theta_t\to\Theta_0=\text{const}$. Training
linearizes; dynamics governed by $\mathrm{spec}(\Theta_0)$.

\vspace{4pt}
\textbf{Identification with SR.} $S_{ij}=\mathrm{Cov}_{|\Psi|^2}[O_i,O_j]$, $O_i=\partial_{\theta_i}\log|\Psi|$.
$S$ is the quantum Fisher matrix = NTK of $\log|\Psi|$ in the sampling measure. SR update
$S\,\delta\theta=-g$ is natural gradient = explicit $TT^\ast$ preconditioner.
\end{frame}

\begin{frame}{Backup: Barron proof sketch}
\small
$g\in L^1(\mathbb R^d)$ with $C_g=\int\|\xi\|_2|\hat g(\xi)|d\xi<\infty$. Write
\[
  g(x)-g(0) \;=\; \int_{\mathbb R^d}(e^{i\xi^\top x}-1)\hat g(\xi)d\xi.
\]
Re-express as $C_g\,\mathbb E_{(\Xi,\tilde\Xi)\sim\mu_g}[\Gamma(\Xi,\tilde\Xi)(x)]$ with $\Gamma$
of fixed form. Average $n$ iid draws; Bienaymé's identity gives variance $\le (2\pi C_g)^2/n$, so the
average is within $2\pi C_g/\sqrt n$ in $L^2(\mu)$. Each $\Gamma$ is well-approximated by a single
neuron. $\Rightarrow$ two-layer net of size $n$ at MC rate.

Monte Carlo samples in frequency space.}}}
\end{frame}
